function confirmExportCSV() { const selectedKey = document.getElementById("exportMonthSelect").value; if (!selectedKey) return; const filteredRecords = records.filter(r => getMonthKey(cleanDate(r.entry)) === selectedKey); if (filteredRecords.length === 0) return alert("Nessun dato per questo mese."); // Intestazione pulita e semplificata let csv = "Data;Giorno;Entrata;Inizio Pausa;Fine Pausa;Uscita;Ore Lavorate;Straordinario;Straordinario Arrotondato\n"; let totalRawMinutes = 0; let totalRawOvertimeMinutes = 0; let totalRoundedOvertimeMinutes = 0; filteredRecords.forEach(r => { let pauseStarts = []; let pauseEnds = []; if (r.pauses && r.pauses.length > 0) { r.pauses.forEach(p => { if (p.start) pauseStarts.push(formatTimeExact(p.start)); if (p.end) pauseEnds.push(formatTimeExact(p.end)); }); } const pStartStr = pauseStarts.join(" | ") || "-"; const pEndStr = pauseEnds.join(" | ") || "-"; const rawWorked = calculateRawWorkedMinutes(r); const rawOvertime = calculateRawOvertime(r); const roundedOvertime = calculateOvertime(r); totalRawMinutes += rawWorked; totalRawOvertimeMinutes += rawOvertime; totalRoundedOvertimeMinutes += roundedOvertime; // Visualizza l'orario reale di entrata/uscita, le ore reali lavorate, lo straordinario reale e quello arrotondato csv += `${formatItalianDate(r.date)};${r.dayName};${formatTimeExact(r.entryExact || r.entry)};${pStartStr};${pEndStr};${formatTimeExact(r.exitExact || r.exit)};${formatDuration(rawWorked)};${formatSignedDuration(rawOvertime)};${formatSignedDuration(roundedOvertime)}\n`; }); // Riga di riepilogo finale csv += `\n`; csv += `;;;;;Totale Mese:;${formatDuration(totalRawMinutes)};${formatSignedDuration(totalRawOvertimeMinutes)};${formatSignedDuration(totalRoundedOvertimeMinutes)}\n`; const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" }); const a = document.createElement("a"); a.href = URL.createObjectURL(blob); a.download = `ore_lavoro_${selectedKey}.csv`; a.click(); closeExportModal(); }
