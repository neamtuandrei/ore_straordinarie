function confirmExportCSV() { const selectedKey = document.getElementById("exportMonthSelect").value; if (!selectedKey) return; const filteredRecords = records.filter(r => getMonthKey(cleanDate(r.entry)) === selectedKey); if (filteredRecords.length === 0) return alert("Nessun dato per questo mese."); // Intestazione tabella giornaliera let csv = "Data;Giorno;Entrata;Pause;Uscita;Ore Lavorate;Straordinario\n"; let monthTotalWorked = 0; let monthTotalOvertime = 0; filteredRecords.forEach(r => { monthTotalWorked += r.workedMinutes; monthTotalOvertime += r.overtimeMinutes; // Formatta ogni intervallo di pausa (es. "12:00-12:30 | 15:00-15:15") let pauseStr = "-"; if (r.pauses && r.pauses.length > 0) { pauseStr = r.pauses.map(p => { const start = formatTime(cleanDate(p.start)); const end = p.end ? formatTime(cleanDate(p.end)) : "In corso"; return `${start}-${end}`; }).join(" | "); } csv += `${formatItalianDate(r.date)};${r.dayName};${formatTime(cleanDate(r.entry))};${pauseStr};${formatTime(cleanDate(r.exit))};${formatDuration(r.workedMinutes)};${formatSignedDuration(r.overtimeMinutes)}\n`; }); // Sezione RIEPILOGO MENSILE in fondo al file CSV csv += `\n`; csv += `RIEPILOGO MENSILE;;;;;;\n`; csv += `Totale Ore Lavorate Mese;;;;;${formatDuration(monthTotalWorked)};\n`; csv += `Totale Straordinario Mese;;;;;;${formatSignedDuration(monthTotalOvertime)}\n`; const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" }); const a = document.createElement("a"); a.href = URL.createObjectURL(blob); a.download = `ore_straordinarie_${selectedKey}.csv`; a.click(); closeExportModal(); }
