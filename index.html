function confirmExportCSV() { const selectedKey = document.getElementById("exportMonthSelect").value; if (!selectedKey) return; const filteredRecords = records.filter(r => getMonthKey(cleanDate(r.entry)) === selectedKey); if (filteredRecords.length === 0) return alert("Nessun dato per questo mese."); const escapeCsv = (val) => { if (val === null || val === undefined) return '""'; const str = String(val); if (str.includes(";") || str.includes('"') || str.includes("\n")) { return `"${str.replace(/"/g, '""')}"`; } return str; }; let csv = "Data;Giorno;Entrata;Inizio Pausa;Fine Pausa;Uscita;Ore Lavorate;Straordinario;Straordinario Arrotondato\n"; let totalRawMinutes = 0; let totalRawOvertimeMinutes = 0; let totalRoundedOvertimeMinutes = 0; }); } const pStartStr = pauseStarts.join(" | ") || "-"; const pEndStr = pauseEnds.join(" | ") || "-"; const rawWorked = calculateRawWorkedMinutes(r); const rawOvertime = calculateRawOvertime(r); const roundedOvertime = calculateOvertime(r); csv += row + "\n"; }); csv += "\n"; csv += `;;;;;Totale Mese:;${escapeCsv(formatDuration(totalRawMinutes))};${escapeCsv(formatSignedDuration(totalRawOvertimeMinutes))};${escapeCsv(formatSignedDuration(totalRoundedOvertimeMinutes))}\n`; const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" }); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = `ore_lavoro_${selectedKey}.csv`;
